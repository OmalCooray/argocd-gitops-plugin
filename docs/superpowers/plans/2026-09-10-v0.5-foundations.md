name: ci

  pull_request:

  contents: read

  test:
    name: pytest (${{ matrix.os }}, py${{ matrix.python }})
    runs-on: ${{ matrix.os }}

        os: [ubuntu-latest, macos-latest, windows-latest]

    name: helm smokes (${{ matrix.os }})
    runs-on: ${{ matrix.os }}
    strategy:
      fail-fast: false
      matrix:
        os: [ubuntu-latest, macos-latest, windows-latest]
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: python -m pip install -r tests/requirements.txt
      - uses: azure/setup-helm@v4
        with:
          version: v3.16.2

        shell: bash
        run: bash tests/smoke/extra_manifest_smoke.sh
      - name: dashboard_smoke
        shell: bash

    name: actionlint

## Task 3: `.github/workflows/e2e.yml` — opt-in kind cluster

**Files:**
- Create: `.github/workflows/e2e.yml`

- [ ] **Step 1: Write `.github/workflows/e2e.yml`**

```yaml
name: e2e

# Slow (~8 min): spins a kind cluster and installs Argo CD. Not on every PR.
on:
  workflow_dispatch:
  push:
    branches: [master]
  pull_request:
    types: [labeled]

permissions:
  contents: read

jobs:
  argocd-e2e:
    name: argocd on kind
    # only run the PR trigger when the 'e2e' label is applied
    if: >-
      github.event_name != 'pull_request' ||
      github.event.label.name == 'e2e'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: python -m pip install -r tests/requirements.txt
      - uses: azure/setup-helm@v4
        with:
          version: v3.16.2
      - uses: helm/kind-action@v1
        with:
          cluster_name: e2e
      - name: argocd_e2e
        shell: bash
        run: bash tests/smoke/argocd_e2e.sh
```

Notes:
- `helm/kind-action@v1` installs `kubectl` and creates the cluster; `argocd_e2e.sh`
  uses the current kube-context, which kind-action sets.
- If `argocd_e2e.sh` assumes a port-forward or a specific storageclass, the
  implementer fixes the script (kind ships `standard` as the default
  storageclass; Argo CD needs no PV). Record any fix.
